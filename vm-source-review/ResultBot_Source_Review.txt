REGEX = re.compile(
    r"^\s*#([A-Z][A-Z0-9&_.-]{0,24})\s*[-–—]\s*"

async def scan():
    env = dotenv_values(ROOT / ".env")
    api_id = int(env.get("TG_API_ID") or 0)
    api_hash = env.get("TG_API_HASH") or ""
    if not api_id or not api_hash:
        raise RuntimeError("Telegram API credentials missing")

    session = env.get("TG_SESSION_STRING")
    if not session:
        data = Path(env.get("DATA_DIR") or "data")
        if not data.is_absolute():
            data = ROOT / data
        session = _encode_sqlite_session_to_string(
            str(data / "tg_userbot.session")
        )

    client = TelegramClient(
        StringSession(session), api_id, api_hash
    )
    await client.connect()
    scanned = positive = intraday = 0

    try:
        if not await client.is_user_authorized():
            raise RuntimeError("Telegram session not authorized")

        channel = await client.get_entity("earnings_pulse")

        async for msg in client.iter_messages(
            channel, limit=2000, wait_time=3
        ):
            scanned += 1
            match = REGEX.match(
                (msg.raw_text or "").split("\n", 1)[0]
            )
            if match:
                positive += 1
                dt = msg.date.astimezone(IST)
                inside = int(
                    dt.weekday() < 5 and
                    Clock(9,15) <=
                    dt.time().replace(tzinfo=None) <
                    Clock(15,30)
                )
                intraday += inside
                DB.execute(
                    "INSERT OR IGNORE INTO alerts VALUES "
                    "(?,?,?,?,?,?,?)",
                    (
                        channel.id, msg.id,
                        match.group(2).upper(),
                        match.group(1).upper(),
                        dt.date().isoformat(),
                        dt.isoformat(), inside
                    )
                )

            if scanned % 500 == 0:
                DB.commit()
                print(
                    "Telegram messages scanned:",
                    scanned, flush=True
                )

        DB.execute(
            "INSERT OR IGNORE INTO flags "
            "VALUES ('scan2000')"
        )
        DB.commit()
        print(
            "Scan complete:", scanned,
            "messages,", positive,
            "positive-rated,", intraday,
            "intraday", flush=True
        )
    finally:
        await client.disconnect()

def yahoo(symbol, days):
    first = (
        datetime.fromisoformat(min(days))
        - timedelta(days=21)
    )
    last = (
        datetime.fromisoformat(max(days))
        + timedelta(days=2)
    )

    url = (
        "https://query2.finance.yahoo.com/v8/finance/chart/"
        + quote(symbol + ".NS", safe="")
    )
    res = requests.get(
        url,
        params={
            "period1":int(first.replace(
                tzinfo=IST).timestamp()),
            "period2":int(last.replace(
                tzinfo=IST).timestamp()),
            "interval":"1d"
        },
        headers={
            "User-Agent":"Mozilla/5.0",
            "Accept":"application/json"
        },
        timeout=25
    )
    if res.status_code in (401,403,429):
        raise RuntimeError(
            "RATE_LIMIT_" + str(res.status_code)
        )
    res.raise_for_status()

    data = (
        res.json().get("chart", {}).get("result")
        or [None]
    )[0]
    if not data:
        raise ValueError("No Yahoo price history")

    meta = data.get("meta", {})
    if (
        meta.get("symbol","").upper() !=
        (symbol+".NS").upper()
        or meta.get("currency") != "INR"
    ):
        raise ValueError("NSE symbol or INR mismatch")

    if meta.get("instrumentType") not in ("EQUITY", None):
        raise ValueError("Not NSE equity")

    prices = {
        datetime.fromtimestamp(
            ts, IST
        ).date().isoformat():float(close)
        for ts, close in zip(
            data.get("timestamp") or [],
            data["indicators"]["quote"][0]["close"]
        )
        if close is not None
        and 0 < float(close) < float("inf")
    }

    return {
        "start":first.date().isoformat(),
        "end":last.date().isoformat(),
        "prices":prices
    }

def report():
    rows = DB.execute(
        "SELECT channel,msg,rating,symbol,"
        "day,stamp,intraday FROM alerts "
        "ORDER BY stamp,msg"
    ).fetchall()

    by_symbol = defaultdict(list)
    for ch,msg,rating,symbol,day,stamp,inside in rows:
        if inside:
            by_symbol[symbol].append(day)

    print(
        "Unique Yahoo NSE symbol lookups:",
        len(by_symbol), flush=True
    )

    prices_by_symbol = {}
    requests_count = success = 0

    for symbol, days in sorted(by_symbol.items()):
        file = CACHE / (symbol.replace("/", "_") + ".json")
        wanted_start = (
            datetime.fromisoformat(min(days))
            - timedelta(days=21)
        ).date().isoformat()
        wanted_end = (
            datetime.fromisoformat(max(days))
            + timedelta(days=2)
        ).date().isoformat()

        cached = None
        if file.exists():
            try:
                cached = json.loads(file.read_text())
            except (ValueError, OSError):
                pass

        if (
            cached
            and cached["start"] <= wanted_start
            and cached["end"] >= wanted_end
        ):
            prices_by_symbol[symbol] = cached["prices"]
            continue

        requests_count += 1
        try:
            item = yahoo(symbol, days)
            file.with_suffix(".tmp").write_text(
                json.dumps(item)
            )
            file.with_suffix(".tmp").replace(file)
            prices_by_symbol[symbol] = item["prices"]
            success += 1
        except Exception as exc:
            print(
                "Yahoo unavailable:",
                symbol, str(exc), flush=True
            )
            if "RATE_LIMIT_" in str(exc):
                print(
                    "Rate limit detected. Stopping "
                    "Yahoo requests, preserving cache.",
                    flush=True
                )
                break

        if requests_count % 15 == 0:
            print(
                "Yahoo requests:", requests_count,
                "successful:", success, flush=True
            )
        time.sleep(3)

    print(
        "Yahoo requests this run:",
        requests_count, "successful:", success
    )

    seen = set()
    stats = defaultdict(list)
    output = []

    for ch,msg,rating,symbol,day,stamp,inside in rows:
        status, prior, close, pct = (
            "OUTSIDE_HOURS","","",""
        )
        if inside:
            status = "MISSING_YAHOO_PRICE"
            bars = prices_by_symbol.get(symbol, {})
            dates = sorted(bars)

            if day in bars:
                pos = dates.index(day)
                if pos > 0:
                    close = bars[day]
                    prior = bars[dates[pos-1]]
                    pct = round(
                        (close/prior-1)*100, 6
                    )
                    status = "MATCHED"
                    key = (rating,symbol,day)

                    if key in seen:
                        status = "DUPLICATE_DAY"
                    else:
                        seen.add(key)
                        stats[rating].append(
                            (symbol,pct)
                        )
                else:
                    status = "PRIOR_CLOSE_MISSING"

        output.append((
            rating,symbol,day,stamp,msg,
            status,prior,close,pct
        ))

    csv_file = STORE / "intraday_closing_report.csv"
    with csv_file.open(
        "w",newline="",encoding="utf-8"
    ) as f:
        w = csv.writer(f)
        w.writerow([
            "rating","symbol","day_ist",
            "alert_ist","message_id","status",
            "previous_close","day_close",
            "daily_change_pct"
        ])
        w.writerows(output)

    print("\nRESULTBOT SAME-DAY CLOSING STUDY")
    print(
        "09:15-15:30 IST only; one stock "
        "per rating per day"
    )
    print(
        f"{'RATING':10} {'ALL':>5} "
        f"{'INTRADAY':>9} {'MATCHED':>8} "
        f"{'UP':>5} {'DOWN':>5} "
        f"{'MEAN%':>9} {'MEDIAN%':>9} {'UP%':>6}"
    )

    for rating in ("EXCELLENT","GREAT","GOOD"):
        vals = [v for _,v in stats[rating]]
        count = sum(
            x[0] == rating for x in output
        )
        within = sum(
            x[0] == rating
            and x[5] != "OUTSIDE_HOURS"
            for x in output
        )
        ups = sum(v>0 for v in vals)
        downs = sum(v<0 for v in vals)

        m = f"{mean(vals):+.2f}" if vals else "N/A"
        med = f"{median(vals):+.2f}" if vals else "N/A"
        up_pct = (
            f"{100*ups/len(vals):.1f}"
            if vals else "N/A"
        )

        print(
            f"{rating:10} {count:5} {within:9} "
            f"{len(vals):8} {ups:5} {downs:5} "
            f"{m:>9} {med:>9} {up_pct:>6}"
        )

    missing = sum(
        x[5] == "MISSING_YAHOO_PRICE"
        for x in output
    )
    print(
        "Intraday alerts without verified "
        "Yahoo prices:", missing
    )
    print("CSV report:", csv_file)
    print(
        "This is FULL-DAY change, NOT gains "
        "after Telegram alert time."
    )

try:

            "Using cached Telegram scan; "

finally:
    DB.close()

GET /profile, GET /orders, GET /trades and GET /orders/{orderId} only.
No placements, modifications, cancellations, token writes or Telegram sessions.
All Telegram messages are queued into the existing ResultBot Log Group outbox.
"""

MAX_LINKS_PER_POLL = 40
TERMINAL = frozenset({"TRADED", "REJECTED", "CANCELLED", "EXPIRED"})
STATUSES = frozenset({"TRANSIT", "PENDING", "PART_TRADED"} | TERMINAL)


def ensure_schema(con: sqlite3.Connection) -> None:
    con.executescript("""
      CREATE TABLE IF NOT EXISTS dhan_order_links (
        id INTEGER PRIMARY KEY,
        channel_id INTEGER NOT NULL,
        telegram_message_id INTEGER NOT NULL,
        dhan_client_id TEXT NOT NULL,
        dhan_order_id TEXT NOT NULL UNIQUE,
        symbol TEXT NOT NULL,
        created_utc INTEGER NOT NULL,
        last_fingerprint TEXT NOT NULL DEFAULT '',
        last_seen_utc INTEGER NOT NULL DEFAULT 0,
        last_broker_status TEXT NOT NULL DEFAULT 'NOT_CHECKED',
        active INTEGER NOT NULL DEFAULT 1 CHECK(active IN (0,1)),
        UNIQUE(channel_id, telegram_message_id)
      );
    """)


def snapshot() -> tuple[str, dict, list[dict], list[dict]]:
    """Read a token afresh each time, so the 08:00 auto-refresh is picked up."""
    account, token = dh.load_credentials()
    profile = dh.verified_profile(account, token)
    orders = dh.verified_rows(dh.get_json("/orders", token), account)
    trades = dh.verified_rows(dh.get_json("/trades", token), account)
    return account, profile, orders, trades


def status_line(order: dict) -> tuple[str, int, int, str]:
    status = str(order.get("orderStatus") or "").upper()
    if status not in STATUSES:
        raise dh.DhanReadError("Dhan returned an unknown or missing order status")
    qty = dh.checked_number(order.get("quantity"))
    filled = dh.checked_number(order.get("filledQty"))
    if qty is None or filled is None or qty <= 0 or filled > qty:
        raise dh.DhanReadError("Dhan returned invalid broker quantities")
    if int(qty) != qty or int(filled) != filled:
        raise dh.DhanReadError("Dhan returned non-integral broker quantities")
    if status == "TRADED" and filled != qty:
        raise dh.DhanReadError("Dhan full-trade status conflicts with filled quantity")
    price = dh.checked_number(order.get("averageTradedPrice"))
    avg = "N/A"
    if filled > 0 and price is not None and price > 0:
        avg = f"₹{price:,.2f}"
    return status, int(filled), int(qty), avg


def validate_identity(order: dict, *, account: str, order_id: str, symbol: str,
                      alert_utc: int | None = None) -> None:
    if str(order.get("orderId") or "") != order_id:
        raise dh.DhanReadError("Broker order ID mismatch")
    if str(order.get("dhanClientId") or "") != account:
        raise dh.DhanReadError("Broker account binding mismatch")
    if str(order.get("tradingSymbol") or "").strip().upper() != symbol:
        raise dh.DhanReadError("Broker symbol mismatch; manual link refused")
    if str(order.get("exchangeSegment") or "") != "NSE_EQ":
        raise dh.DhanReadError("Only verified NSE equity orders can link to Excellent equity alerts")
    if str(order.get("transactionType") or "") != "BUY":
        raise dh.DhanReadError("Only verified BUY orders can link to entry alerts")
    status_line(order)
    if alert_utc is not None:
        try:
            created = datetime.strptime(str(order.get("createTime") or ""), "%Y-%m-%d %H:%M:%S")
            created_utc = int(created.replace(tzinfo=watch.IST).timestamp())
        except ValueError:
            raise dh.DhanReadError("Broker order createTime missing or invalid")
        if created_utc < int(alert_utc) - 5:
            raise dh.DhanReadError("Broker order predates the Telegram alert; link refused")
        if created_utc > time.time() + 60:
            raise dh.DhanReadError("Broker order timestamp is in the future; link refused")


def queue_message(con: sqlite3.Connection, *, telegram_id: int, kind: str,
                  body: str) -> bool:
    if not re.fullmatch(r"[A-Z0-9_]{3,60}", kind):
        raise ValueError("Unsafe notification kind")
    cur = con.execute(
        """INSERT OR IGNORE INTO log_outbox
        (channel_id,message_id,kind,text_html,created_utc)
        VALUES (?,?,?,?,?)""",
        (watch.CHANNEL_ID, int(telegram_id), kind, body, int(time.time())),
    )
    return bool(cur.rowcount)


def connect(con: sqlite3.Connection, *, notify: bool = True) -> dict:
    """Live API verification; queue exactly one non-execution confirmation card."""
    ensure_schema(con)
    account, profile, orders, trades = snapshot()
    if notify:
        card = "\n".join((
            "🏦 <b>RESULTBOT — DHAN CONNECTED</b>",
            f"✅ Verified Dhan account: <b>{escape(dh.display_account(account))}</b>",
            "🔐 Token refresh: separate daily systemd timer",
            f"📋 Today's Dhan orders: {len(orders)}",
            f"📒 Today's Dhan trades: {len(trades)}",
            "🟡 Mode: READ-ONLY — order placement DISABLED",
            "🔗 No trade is attributed to an Excellent alert without a verified order link",
        ))
        queue_message(con, telegram_id=0, kind="DHAN_CONNECTED_V1", body=card)
    return {"account": dh.display_account(account),
            "token_validity": str(profile.get("tokenValidity") or "Not supplied"),
            "orders_today": len(orders), "trades_today": len(trades)}


def recent_alerts(con: sqlite3.Connection, limit: int = 10) -> None:
    if not 1 <= limit <= 30:
        raise dh.DhanReadError("Limit must be 1-30")
    print("Verified Earnings Pulse Excellent alerts (real source only):")
    for message_id, symbol, sent in con.execute(
        """SELECT message_id,symbol,sent_utc FROM alerts WHERE channel_id=?
           ORDER BY sent_utc DESC LIMIT ?""",(watch.CHANNEL_ID,limit)
    ):
        when = datetime.fromtimestamp(int(sent),watch.IST).strftime("%d-%b %H:%M IST")
        print(f"  Telegram message {message_id} | #{symbol} | {when}")
    print("These are captured alerts, not broker orders.")


def link_order(con: sqlite3.Connection, *, telegram_id: int, order_id: str) -> str:
    """Only explicit owner-supplied broker order ID can create a mapping."""
    if not re.fullmatch(r"[0-9]{1,25}", order_id or ""):
        raise dh.DhanReadError("Broker order ID must be numeric")
    if not isinstance(telegram_id,int) or telegram_id <= 0:
        raise dh.DhanReadError("Telegram alert message ID must be positive")
    ensure_schema(con)
    alert = con.execute(
        """SELECT symbol,sent_utc FROM alerts
           WHERE channel_id=? AND message_id=?""",
        (watch.CHANNEL_ID,telegram_id)
    ).fetchone()
    if not alert:
        raise dh.DhanReadError("Telegram ID is not a stored REAL Earnings Pulse Excellent alert")
    symbol, sent_utc = str(alert[0]), int(alert[1])
    account, token = dh.load_credentials()
    dh.verified_profile(account, token)
    order = dh.get_json("/orders/"+order_id, token)
    if not isinstance(order, dict):
        raise dh.DhanReadError("Dhan order response must be an object")
    validate_identity(order, account=account,order_id=order_id,
                      symbol=symbol,alert_utc=sent_utc)
    try:
        con.execute(
            """INSERT INTO dhan_order_links
               (channel_id,telegram_message_id,dhan_client_id,dhan_order_id,symbol,created_utc)
               VALUES (?,?,?,?,?,?)""",
            (watch.CHANNEL_ID,telegram_id,account,order_id,symbol,int(time.time()))
        )
    except sqlite3.IntegrityError:
        existing = con.execute(
            """SELECT telegram_message_id,dhan_order_id FROM dhan_order_links
               WHERE dhan_order_id=? OR (channel_id=? AND telegram_message_id=?)""",
            (order_id,watch.CHANNEL_ID,telegram_id)
        ).fetchone()
        if existing == (telegram_id,order_id):
            return "ALREADY_LINKED"
        raise dh.DhanReadError("Order ID or alert already linked; refusing ambiguous attribution")
    return "LINKED"


def _trade_summary(trades: list[dict], order_id: str) -> tuple[int, str]:
    """Weighted actual fill price from trade records with matching order ID."""
    qty=0
    total=0.0
    for trade in trades:
        if str(trade.get("orderId") or "")!=order_id:
            continue
        count=dh.checked_number(trade.get("tradedQuantity"))
        price=dh.checked_number(trade.get("tradedPrice"))
        if count is None or int(count)!=count or count<=0 or price is None or price<=0:
            raise dh.DhanReadError("Malformed broker trade quantity or price")
        qty+=int(count)
        total+=int(count)*price
    return qty, f"₹{total/qty:,.2f}" if qty else "N/A"


def apply_snapshot(con: sqlite3.Connection, *, account: str, orders: list[dict],
                   trades: list[dict]) -> int:
    """Queue only changed statuses of explicitly linked real-source orders."""
    ensure_schema(con)
    # Account mismatch from the cached link is a hard stop, never a symbol match.
    links = con.execute(
        """SELECT id,telegram_message_id,dhan_client_id,dhan_order_id,symbol,last_fingerprint
           FROM dhan_order_links WHERE active=1 ORDER BY id LIMIT ?""",
        (MAX_LINKS_PER_POLL,)
    ).fetchall()
    if not links:
        return 0
    order_map={str(x.get("orderId")):x for x in orders}
    changed=0
    for link_id, message_id, bound_account, order_id, symbol, old_fingerprint in links:
        if bound_account!=account:
            raise dh.DhanReadError("Saved broker account differs from current Dhan account")
        order=order_map.get(order_id)
        if order is None:
            # Dhan order API is current-day only. Do not claim order rejection.
            continue
        validate_identity(order,account=account,order_id=order_id,symbol=symbol)
        status, filled, qty, avg = status_line(order)
        trade_qty, trade_avg = _trade_summary(trades,order_id)
        if trade_qty>qty or trade_qty>filled:
            raise dh.DhanReadError("Broker trade quantities disagree with order")
        if filled and trade_qty == filled and trade_avg != "N/A":
            reported_price = trade_avg
            source = "Dhan trade book"
        elif filled and avg != "N/A":
            reported_price = avg
            source = "Dhan order book"
        else:
            reported_price = "N/A"
            source = "not confirmed"
        fingerprint = json.dumps([status,filled,qty,reported_price,source])
        if fingerprint==old_fingerprint:
            continue
        digest=hashlib.sha256(fingerprint.encode()).hexdigest()[:20].upper()
        body="\n".join((
            "🏦 <b>RESULTBOT — DHAN BROKER UPDATE</b>",
            f"📈 Excellent alert: <b>#{escape(symbol)}</b>",
            f"📩 Telegram message: {int(message_id)}",
            f"📌 Dhan order status: <b>{escape(status)}</b>",
            f"✅ Filled quantity: {filled}/{qty}",
            f"💰 Verified average execution: <b>{escape(reported_price)}</b>",
            f"🔎 Price source: {escape(source)}",
            "🔗 Order manually linked and broker-verified",
            "🟡 Order NOT submitted by ResultBot; monitoring only",
        ))
        queue_message(con,telegram_id=int(message_id),kind="DHAN_"+digest,body=body)
        con.execute(
            """UPDATE dhan_order_links
               SET last_fingerprint=?,last_seen_utc=?,last_broker_status=?,active=?
               WHERE id=?""",
            (fingerprint,int(time.time()),status,0 if status in TERMINAL else 1,link_id)
        )
        changed+=1
    return changed


def poll_once(con: sqlite3.Connection) -> int:
    """Synchronous manual poll of matched orders, no action when none linked."""
    ensure_schema(con)
    count=con.execute("SELECT COUNT(*) FROM dhan_order_links WHERE active=1").fetchone()[0]
    if not count:
        return 0
    if count>MAX_LINKS_PER_POLL:
        raise dh.DhanReadError("Too many active broker links; manual review required")
    account, _, orders, trades = snapshot()
    return apply_snapshot(con,account=account,orders=orders,trades=trades)


async def poll_async(con: sqlite3.Connection) -> int:
    """Perform slow HTTP in worker, SQLite writes on Telegram listener thread."""
    ensure_schema(con)
    count=con.execute("SELECT COUNT(*) FROM dhan_order_links WHERE active=1").fetchone()[0]
    if not count:
        return 0
    if count>MAX_LINKS_PER_POLL:
        raise dh.DhanReadError("Too many active broker links; manual review required")
    account,_,orders,trades = await asyncio.to_thread(snapshot)
    return apply_snapshot(con,account=account,orders=orders,trades=trades)

def get_json(path: str, token: str) -> object:
    if not validate_token(token):
        raise DhanReadError("Access token invalid or missing")
    url = safe_endpoint(path)
    req = Request(url, method="GET", headers={
        "access-token": token, "Accept": "application/json",
        "User-Agent": "ResultBot-ReadOnly/1.0",
    })
    try:
        with urlopen(req, timeout=12) as response:
            if response.status != 200:
                raise DhanReadError("Dhan returned an unexpected status")
            raw = response.read(2_000_001)
            if len(raw) > 2_000_000:
                raise DhanReadError("Dhan response exceeds size limit")
            result = json.loads(raw)
            if isinstance(result, dict) and result.get("errorCode"):
                raise DhanReadError("Dhan returned an API error")
            return result
    except HTTPError as exc:
        if exc.code in (401,403):
            raise DhanReadError("Authentication rejected or token expired (HTTP %d)" % exc.code)
        raise DhanReadError("Dhan HTTP error: %d" % exc.code)
    except (URLError, TimeoutError, OSError):
        raise DhanReadError("Dhan connection failed; check network/API availability")
    except (ValueError, UnicodeDecodeError):
        raise DhanReadError("Dhan returned invalid JSON")

def verify_fund_limits(account: str, response: object) -> dict[str, float | None]:
    """Fail closed on account mismatch and malformed balances; never infer funds.

    The GET /fundlimit response is not guaranteed to match available cash that
    can be used for any particular order; broker margin/risk checks still apply.
    """
    if not valid_account(account):
        raise DhanReadError("Invalid account provided for balance verification")
    if not isinstance(response, dict):
        raise DhanReadError("Dhan fund-limit response has invalid structure")
    if str(response.get("dhanClientId") or "") != account:
        raise DhanReadError("Dhan funds account mismatch; refusing balance")
    verified: dict[str, float | None] = {}
    for key, _ in FUND_FIELDS:
        value = response.get(key)
        if value is None:
            if key == "availabelBalance":
                raise DhanReadError(
                    "Dhan did not provide its required availabelBalance field"
                )
            verified[key] = None
            continue
        # Explicitly reject bool and values that cannot be trusted as money.
        if isinstance(value, bool):
            raise DhanReadError("Dhan returned an invalid numeric balance")
        amount = checked_number(value)
        if amount is None:
            raise DhanReadError("Dhan returned a negative or invalid fund-limit amount")
        verified[key] = amount
    return verified

class _NeverRedirect(HTTPRedirectHandler):
    """Do not follow auth redirects carrying transient PIN/TOTP query values."""
    def redirect_request(self, req, fp, code, msg, headers, newurl):

def current_totp_from_secret(secret: str, *, timestamp: float | None = None) -> str:
    """Convert a TOTP setup key to the CURRENT 6-digit RFC 6238 code in memory.

    Neither the setup secret nor generated code is persisted, logged or printed.
    Only standard 30-second HMAC-SHA1 TOTP is supported.
    """
    if not isinstance(secret, str):
        raise DhanReadError("TOTP setup key must be text")
    clean = "".join(secret.upper().split()).replace("-", "")
    if not 16 <= len(clean) <= 128 or not re.fullmatch(r"[A-Z2-7]+", clean):
        raise DhanReadError(
            "Invalid TOTP setup key: use the 16-character Base32 key, "
            "or the current 6-digit code from your authenticator"
        )
    padded = clean + ("=" * ((-len(clean)) % 8))
    try:
        key = base64.b32decode(padded, casefold=True)
    except (binascii.Error, ValueError) as exc:
        raise DhanReadError("TOTP setup key could not be decoded") from None
    if len(key) < 10:
        raise DhanReadError("TOTP setup key is too short")
    now = time.time() if timestamp is None else timestamp
    try:
        counter = int(now // 30)
        digest = hmac.new(key, struct.pack(">Q",counter), hashlib.sha1).digest()
    except (TypeError,ValueError,OverflowError):
        raise DhanReadError("Invalid time for TOTP code generation") from None
    offset = digest[-1] & 0x0F
    truncated = struct.unpack(">I",digest[offset:offset+4])[0] & 0x7fffffff
    return str(truncated % 1000000).zfill(6)


def normalize_totp_input(input_value: str) -> str:
    """Accept either current 6 digits or a secret key; do not retain secret."""
    candidate = str(input_value or "").strip()
    if re.fullmatch(r"[0-9]{6}",candidate):
        return candidate
    return current_totp_from_secret(candidate)


def token_from_pin_totp(account: str, pin: str, totp: str) -> str:
    """Generate a 24-hour Dhan token; do NOT store PIN, TOTP or TOTP seed.

    This is the only non-GET call in this module. Dhan documents these
    parameters in a POST HTTPS query string, so do not log the URL.
    """
    if not valid_account(account):
        raise DhanReadError("Invalid Dhan Client ID")
    if not re.fullmatch(r"[0-9]{6}", pin or ""):
        raise DhanReadError("Dhan PIN must contain exactly 6 digits")
    if not re.fullmatch(r"[0-9]{6}", totp or ""):
        raise DhanReadError("TOTP must contain exactly 6 digits")
    url = "https://auth.dhan.co/app/generateAccessToken?" + urlencode({
        "dhanClientId": account,
        "pin": pin,
        "totp": totp,
    })
    request = Request(url, data=b"", method="POST", headers={
        "Accept":"application/json",
        "User-Agent":"ResultBot-Dhan-Auth/1.0",
    })
    try:
        with build_opener(_NeverRedirect()).open(request, timeout=12) as reply:
            if reply.status != 200:
                raise DhanReadError("Dhan token generation returned an unexpected status")
            raw = reply.read(200_001)
            if len(raw) > 200_000:
                raise DhanReadError("Dhan authentication response exceeded size limit")
            data = json.loads(raw)
    except HTTPError as exc:
        # Never expose exc.url, which contains the PIN and TOTP.
        raise DhanReadError("Dhan TOTP authentication rejected (HTTP %d). Check PIN, current code and clock." % exc.code) from None
    except (URLError, TimeoutError, OSError):
        raise DhanReadError("Dhan authentication connection failed; check connectivity") from None
    except (ValueError, UnicodeDecodeError):
        raise DhanReadError("Dhan authentication returned invalid JSON") from None
    if not isinstance(data, dict):
        raise DhanReadError("Dhan authentication returned an unexpected response")
    if str(data.get("dhanClientId") or "") != account:
        raise DhanReadError("Dhan generated a token for a different client ID; refusing it")
    token = data.get("accessToken")
    if not validate_token(token):
        raise DhanReadError("Dhan authentication did not return a valid access token")
    return token



@contextmanager
def renewal_lock(path: Path = RENEW_LOCK):
    """Prevent two refresh invocations from generating competing credentials."""
    path.parent.mkdir(parents=True, exist_ok=True, mode=0o700)
    path.parent.chmod(0o700)
    safe_path(path)
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_NOFOLLOW, 0o600)
    try:
        os.fchmod(fd, 0o600)
        fcntl.flock(fd, fcntl.LOCK_EX)
        yield
    finally:
        fcntl.flock(fd, fcntl.LOCK_UN)
        os.close(fd)

@contextmanager
def managed_db(path: Path | None = None):
    """Commit-safe, explicitly closed SQLite handle (no leaked broker state handles)."""
    con=connect_db(path)
    try:
        yield con
    finally:
        con.close()


def _control(con) -> dict:
    row=con.execute(
        "SELECT armed,source,budget,expires,armed_since,arm_seq,stop_seq FROM control WHERE id=1"
    ).fetchone()
    return dict(zip(("armed","source","budget","expires","armed_since","arm_seq","stop_seq"),row))


def status() -> dict:
    with managed_db() as con:
        ctrl=_control(con)
        rows=con.execute("""SELECT id,symbol,state,order_id,broker_status,qty,price,
             created_ts,note,cancel_after FROM intents ORDER BY id DESC""").fetchall()
        ctrl["orders"]=[dict(zip(
            ("id","symbol","state","order_id","broker_status","qty","price",
             "created_ts","note","cancel_after"),r)) for r in rows]
        ctrl["effective_on"]=bool(ctrl["armed"])
        ctrl["active_orders"]=con.execute(
            "SELECT COUNT(*) FROM intents WHERE state IN "
            "('RESERVED','PREPARING','SUBMITTING','UNKNOWN','ACKED','CANCEL_REQUESTED')"
        ).fetchone()[0]
        return ctrl

def update_settings(*, source: str | None = None,
                    budget: int | None = None) -> dict:

    This never contacts Dhan, submits or cancels an order, or restarts services.

        raise LiveAmoError("Specify --source or --budget to edit settings")
    if source is not None and source not in ("TEST", "EARNINGS_PULSE", "BOTH"):
        raise LiveAmoError("Source must be TEST, EARNINGS_PULSE or BOTH")

            if source is not None and next_source != c["source"]:

def reserve(source: str, msg_id: int, symbol: str, sent_utc: int) -> int | None:
    """Persistent ON gate: each distinct fresh Telegram message gets one durable intent."""
    if source not in ("TEST","EARNINGS_PULSE") or not re.fullmatch(
        r"[A-Z][A-Z0-9&._-]{0,29}",symbol or ""
    ):
        return None
    now=int(time.time())
    if not 0<=now-int(sent_utc)<=MAX_SIGNAL_AGE:
        return None
    if not trading_service_stopped():
        disarm()
        return None
    with managed_db() as con:
        con.execute("BEGIN IMMEDIATE")
        try:
            c=_control(con)
            if not(c["armed"] and c["source"] in (source,"BOTH")
                    and int(sent_utc)>=c["armed_since"]):
                con.execute("COMMIT")
                return None
            # NO auto-disarm. UNIQUE(source,msg_id) ensures at-most-one intent
            # per Telegram message across restarts. OFF is owner-controlled.
            account,_=dh.load_credentials()
            # Distinguish overlapping message IDs from the two sources.
            # Dhan correlation IDs must fit 30 characters and are never reused
            # for distinct Telegram messages, including after switch toggles.
            digest=hashlib.sha256(
                f"{source}:{msg_id}:{c['arm_seq']}".encode("utf-8")
            ).hexdigest()[:24]
            correlation="RBA-"+digest
            cur=con.execute(
                """INSERT OR IGNORE INTO intents
                 (source,msg_id,symbol,alert_ts,created_ts,account,arm_stop_seq,budget,correlation)
                 VALUES(?,?,?,?,?,?,?,?,?)""",
                 (source,int(msg_id),symbol,int(sent_utc),now,account,c["stop_seq"],c["budget"],correlation)
            )
            con.execute("COMMIT")
            return int(cur.lastrowid) if cur.rowcount else None
        except BaseException:
            con.execute("ROLLBACK")
            raise

def _dhan_http(method:str,path:str,token:str,payload:dict|None=None) -> dict:
    if method=="POST" and path!="/orders":
        raise LiveAmoError("Only Dhan POST /orders is authorized for entry")
    if method=="DELETE" and not re.fullmatch(r"/orders/[0-9]{6,25}",path):
        raise LiveAmoError("Only explicit resultbot-owned Dhan order cancellation allowed")
    if method not in ("POST","DELETE"):
        raise LiveAmoError("Only placement/cancellation broker mutations supported")
    body=json.dumps(payload).encode() if payload is not None else None
    req=Request("https://api.dhan.co/v2"+path,data=body,method=method,headers={
      "access-token":token,"Content-Type":"application/json","Accept":"application/json",
      "User-Agent":"ResultBot-Amo-OwnerOptIn/1.0",
    })
    # No retries. Ambiguous timeouts always require correlation reconciliation.
    try:
        with urlopen(req,timeout=8) as response:
            if response.status not in (200,201,202):
                raise LiveAmoError("Unexpected Dhan HTTP status; manually reconcile")
            data=json.loads(response.read(150_000))
        if not isinstance(data,dict):
            raise LiveAmoError("Broker mutation response is not an object")
        return data
    except (HTTPError,URLError,TimeoutError,OSError,ValueError):
        raise LiveAmoError("Broker outcome UNKNOWN or rejected; reconcile by correlation/order ID") from None


def _egress_ip() -> str:
    try:
        with urlopen(Request(EGRESS_CHECK_URL,headers={"User-Agent":"ResultBot-IP/1.0"}),timeout=6) as r:
            ip=r.read(100).decode().strip()
    except (URLError,OSError,TimeoutError):
        raise LiveAmoError("Cannot verify VM outbound IP") from None
    if not re.fullmatch(r"[0-9]{1,3}(\.[0-9]{1,3}){3}",ip):
        raise LiveAmoError("Invalid outbound IP response")
    return ip


def _fresh_dhan_funds(account: str, token: str) -> float:
    """Strictly read-only funds retry for temporary Dhan HTTP 5xx.

    Never use a remembered balance, and never retry a broker order POST.
    """
    for attempt in range(3):
        try:
            data=dh.get_json("/fundlimit",token)
            validated=dh.verify_fund_limits(account,data)
            return float(validated["availabelBalance"])
        except dh.DhanReadError as exc:
            msg=str(exc)
            transient=any(f"Dhan HTTP error: {code}" in msg
                          for code in (500,502,503,504))
            if not transient:
                raise
            if attempt==2:
                raise LiveAmoError(
                    "Dhan GET /fundlimit returned HTTP 5xx after 3 "
                    "GET-only attempts; live balance unverified. "
                    "AMO remains OFF; do not submit a real order."
                ) from None
            time.sleep((attempt+1)*1.0)
    raise LiveAmoError("Dhan funds unavailable; AMO blocked")


def _verified_broker_context(account:str,token:str):
    dh.verified_profile(account,token)
    ip=dh.get_json("/ip/getIP",token)
    if not isinstance(ip,dict):
        raise LiveAmoError("Dhan registered-IP response invalid")
    outbound=_egress_ip()
    if outbound not in (ip.get("primaryIP"),ip.get("secondaryIP")):
        raise LiveAmoError("ResultBot outbound IP is not on the Dhan order whitelist")
    # Live funds must always be freshly verified; a temporary upstream
    # HTTP 500 is retried ONLY for the safe, idempotent GET endpoint.
    return _fresh_dhan_funds(account,token)


def _session_closed() -> bool:
    now=datetime.now(IST)

def _quote_and_size(symbol:str,budget:int,tick:Decimal):
    bars=watch.request_bars(symbol)
    if not bars:
        raise LiveAmoError("No independently verified Yahoo NSE price bars")
    last=max(bars,key=lambda b:int(b["ts_utc"]))
    age=time.time()-int(last["ts_utc"])
    if not 60<=age<=72*3600:
        raise LiveAmoError("Latest market reference is missing or older than 72 hours")
    # 0.25% cushion, rounded UP to a verified instrument tick.
    close=Decimal(str(last["close"]))
    limit=(close*Decimal("1.0025")/tick).to_integral_value(rounding=ROUND_CEILING)*tick
    if limit <= 0 or limit >= close * Decimal("1.03"):
        raise LiveAmoError("Quoted AMO limit is invalid")
    qty=int(Decimal(budget)//limit)
    if qty<=0 or Decimal(qty)*limit > budget:
        raise LiveAmoError("Stock too expensive for current ₹"+str(budget)+" budget")
    return qty,float(limit),int(last["ts_utc"])

def request_cancel(intent_id:int,automatic:bool=False) -> str:
    """Cancel only recorded ResultBot-owned pending order; never assume success."""
    item=_intent(intent_id)
    if item["state"] not in ("ACKED","UNKNOWN","CANCEL_REQUESTED"):
        raise LiveAmoError("Cannot safely cancel: no active resultbot-owned order")
    if not re.fullmatch(r"[0-9]{6,25}",item["order_id"]):
        raise LiveAmoError("Cannot cancel without verified Dhan order ID")
    account, token=dh.load_credentials()
    if account!=item["account"]:
        raise LiveAmoError("Dhan account mismatch before cancellation")
    detail=_verify_owned_order(item,token)
    status=str(detail.get("orderStatus") or "").upper()
    filled=dh.checked_number(detail.get("filledQty"))
    if status in TERMINAL:
        _update(intent_id,"TERMINAL",
                broker_status=status,note="Order already terminal; cannot cancel")
        return status
    # PART_TRADED still has a cancellable pending remainder; an existing
    # filled portion cannot be undone and must be managed separately.
    if status not in ("TRANSIT","PENDING","PART_TRADED"):
        raise LiveAmoError("Unrecognized broker state; cancellation blocked")
    # Lock and claim ONE cancellation attempt only after broker preflight succeeds.
    with managed_db() as con:
        claim=con.execute(
            "UPDATE intents SET cancel_attempted=1 WHERE id=? AND cancel_attempted=0",
            (intent_id,)
        )
        if claim.rowcount != 1:
            raise LiveAmoError("Cancel already attempted; check Dhan order book")
    try:
        result=_dhan_http("DELETE","/orders/"+item["order_id"],token)
    except LiveAmoError:
        _update(intent_id,"CANCEL_REQUESTED",note="Cancellation outcome uncertain; check Dhan directly")
        return "CANCEL_UNCONFIRMED"
    result_status=str(result.get("orderStatus") or "TRANSIT").upper()
    if str(result.get("orderId"))!=item["order_id"]:
        _update(intent_id,"CANCEL_REQUESTED",note="Cancellation response mismatch; check Dhan")
        return "CANCEL_UNCONFIRMED"
    _update(intent_id,"CANCEL_REQUESTED",broker_status=result_status,
            note="Dhan cancel request received; awaiting broker status")
    return result_status


def _order_by_correlation(correlation: str, token: str) -> dict | None:
    """Dhan GET /orders/external/{correlationId}; never submit twice on ambiguity."""
    if not re.fullmatch(r"[a-zA-Z0-9_-]{3,30}",correlation):
        raise LiveAmoError("Unsafe correlation ID")
    req=Request("https://api.dhan.co/v2/orders/external/"+correlation,
                headers={"access-token":token,"Accept":"application/json"},
                method="GET")
    try:
        with urlopen(req,timeout=10) as r:
            payload=json.loads(r.read(200_000))
    except HTTPError as exc:
        if exc.code==404:
            return None
        raise LiveAmoError("Cannot reconcile Dhan order by correlation ID") from None
    except (URLError,OSError,TimeoutError,ValueError):
        raise LiveAmoError("Broker reconciliation temporarily unavailable") from None
    if not isinstance(payload,dict):
        raise LiveAmoError("Unexpected Dhan correlation response")
    return payload


def reconcile_unknown(intent_id:int) -> bool:

        raise LiveAmoError("Dhan correlation mismatch")

        raise LiveAmoError("Dhan correlation returned invalid order ID")

        raise LiveAmoError("Cannot verify order instrument from uncertain submission")

def monitor() -> int:
    """Monitor ALL resultbot-owned live orders; cancellation is owner-controlled."""

        )]

                    changed += 1
            except (LiveAmoError,dh.DhanReadError,OSError,ValueError):
                pass

            order=_verify_owned_order(item,token)
            status=str(order.get("orderStatus") or "").upper()
            filled=dh.checked_number(order.get("filledQty"))
            if status not in ("TRANSIT","PENDING","PART_TRADED","TRADED","REJECTED","CANCELLED","EXPIRED"):
                raise LiveAmoError("Unknown broker order status")
            if status!=item["broker_status"] or (filled is not None and filled>0):
                _update(intent_id,"TERMINAL" if status in TERMINAL else "ACKED",
                        broker_status=status,note="Broker-verified order update")

            # chooses which broker-confirmed orders to cancel and when.
        except (LiveAmoError,dh.DhanReadError,ValueError,OSError):
            # State remains unresolved and should be reviewed in Dhan.
            continue
    return changed

async def send_queued_logs(con: sqlite3.Connection, client, group) -> int:
    """Crash-recoverable at-least-once delivery; no historical import notifications."""
    if group is None:
        return 0
    now=int(time.time())
    # A previous process may have crashed after claim. Retrying could deliver a
    # duplicate in an ambiguous network failure; never claim exactly once.
    con.execute(
        "UPDATE log_outbox SET state='PENDING' WHERE state='SENDING' AND last_try_utc<?",
        (now-600,),
    )
    rows=con.execute(
        """SELECT id,text_html,created_utc FROM log_outbox
           WHERE state='PENDING' AND attempts<5 AND last_try_utc<=?
           ORDER BY id LIMIT 10""",
        (now-30,),
    ).fetchall()
    delivered=0
    for rowid, body, created in rows:
        if created < now-24*3600:
            con.execute("UPDATE log_outbox SET state='EXPIRED' WHERE id=?",(rowid,))
            continue
        claimed=con.execute(
            """UPDATE log_outbox SET state='SENDING',attempts=attempts+1,last_try_utc=?
               WHERE id=? AND state='PENDING'""",
            (now,rowid),
        )
        if not claimed.rowcount:
            continue
        try:
            await client.send_message(group,body,parse_mode="html",link_preview=False)
        except Exception as exc:
            con.execute("UPDATE log_outbox SET state='PENDING' WHERE id=?",(rowid,))
            LOG.warning("Excellent Log Group delivery failed: %s",type(exc).__name__)
        else:
            con.execute(
                "UPDATE log_outbox SET state='SENT',sent_utc=? WHERE id=?",
                (int(time.time()),rowid),
            )
            delivered+=1
    return delivered


def request_bars(symbol: str, timeout: int = 12) -> list[dict]:
    """Return *completed*, provider-timestamped NSE 1m candles, or fail closed."""
    if not re.fullmatch(r"[A-Z][A-Z0-9&._-]{0,29}", symbol):
        raise ValueError("Invalid symbol")
    ticker = symbol + ".NS"
    query = urlencode({"interval": "1m", "range": "5d"})
    url = f"https://query1.finance.yahoo.com/v8/finance/chart/{quote(ticker, safe='')}?{query}"
    req = Request(url, headers={"User-Agent": "Mozilla/5.0 ResultBot-Research/1.0", "Accept": "application/json"})
    with urlopen(req, timeout=timeout) as reply:
        data = json.load(reply)
    chart = data.get("chart") or {}
    results = chart.get("result") or []
    if not results:
        raise ValueError(f"No Yahoo chart for {ticker}: {str(chart.get('error'))[:180]}")
    result = results[0]
    meta = result.get("meta") or {}
    returned = str(meta.get("symbol", "")).upper()
    if returned != ticker.upper() or meta.get("currency") != "INR":
        raise ValueError(f"Yahoo identity mismatch: asked={ticker}, returned={returned}, currency={meta.get('currency')}")
    quotes = (result.get("indicators") or {}).get("quote") or []
    if not quotes:
        raise ValueError("Yahoo returned no quote arrays")
    quoted = quotes[0]
    out = []
    now = time.time()
    for i, ts in enumerate(result.get("timestamp") or []):
        try:
            values = {k: float(quoted[k][i]) for k in FIELDS}
            if not all(math.isfinite(v) and v > 0 for v in values.values()):
                continue
            ts = int(ts)
            # Exclude not-yet-closed minute to prevent use of moving candles.
            if ts + 60 > now + 1:
                continue
            ist = datetime.fromtimestamp(ts, IST)
            if ist.weekday() >= 5 or not (9 * 60 + 15 <= ist.hour*60 + ist.minute < 15*60 + 30):
                continue
            if values["high"] < values["low"] or not values["low"] <= values["close"] <= values["high"]:
                continue
            v = quoted.get("volume") or []
            vol = v[i] if i < len(v) and isinstance(v[i], (float, int)) and v[i] >= 0 else None
            out.append({"symbol": symbol, "ts_utc": ts, **values,
                        "volume": int(vol) if vol is not None else None, "fetched_utc": int(now)})
        except (TypeError, ValueError, KeyError, IndexError):
            continue
    return out


def store_bars(con: sqlite3.Connection, symbol: str, bars: list[dict]) -> int:

def track_once(con: sqlite3.Connection, *, now: int | None = None) -> tuple[int, int]:
    """Only request symbols with recent Excellent alerts. Not all market symbols."""
    now = int(now or time.time())
    earliest = now - 5 * 86400
    symbols = [row[0] for row in con.execute("SELECT DISTINCT symbol FROM alerts WHERE sent_utc>=?", (earliest,))]
    worked = 0
    for symbol in symbols:
        # Respect free-data endpoint and give delayed minute bars time to settle.
        last = con.execute("SELECT checked_utc FROM fetch_status WHERE symbol=?", (symbol,)).fetchone()
        if last and now - int(last[0]) < 180:
            continue
        try:
            bars = request_bars(symbol)
            n = store_bars(con, symbol, bars)
            con.execute("REPLACE INTO fetch_status VALUES (?,?,?,?)", (symbol, "OK" if n else "NO_BARS", now, f"candles={n}"))
            worked += 1
        except (HTTPError, URLError, OSError, ValueError, TimeoutError) as e:
            con.execute("REPLACE INTO fetch_status VALUES (?,?,?,?)", (symbol, "UNAVAILABLE", now, str(e)[:250]))
            LOG.warning("Price unavailable for %s: %s", symbol, str(e)[:160])
    return worked, len(symbols)

def summarize(con: sqlite3.Connection, alert_id: int) -> dict:
    row = con.execute("SELECT symbol, sent_utc, message_id FROM alerts WHERE id=?", (alert_id,)).fetchone()
    if not row:
        raise ValueError("Missing alert")
    sym, alert_ts, message_id = row
    # A 1-minute bar spans [ts, ts+60); require open >= alert time, and
    # strictly later than the alert's minute (pre-alert trades may share that minute).
    first_possible = (alert_ts//60 + 1)*60
    first = con.execute("""SELECT ts_utc,close FROM one_minute_bars WHERE symbol=? AND ts_utc>=?
      AND ts_utc<=? ORDER BY ts_utc LIMIT 1""", (sym, first_possible, alert_ts + 5*86400)).fetchone()
    d = {"symbol": sym, "message_id": message_id, "alert_utc": alert_ts,
         "first_bar_utc": None, "first_price": None, "status": "NO_VERIFIED_POST_ALERT_CANDLE"}
    if first is None:
        return d
    first_ts, first_price = first
    d.update(first_bar_utc=first_ts, first_price=first_price, status="VERIFIED_1M_CANDLE")
    # Only same market session, only first 60 minutes after first price observation.
    rows = [r for r in con.execute("""SELECT ts_utc,low,high,close FROM one_minute_bars
      WHERE symbol=? AND ts_utc>=? AND ts_utc<=? ORDER BY ts_utc""",
      (sym, first_ts, first_ts+3600)) if same_session(first_ts, r[0])]
    # The reference first_price is the CLOSE of first_ts: its own high/low
    # could have happened BEFORE that close and must never be counted as
    # post-reference excursions. Only LATER completed bars are eligible.
    range_rows = [r for r in rows if r[0] > first_ts]
    d["observed_low_pct"] = round((min(r[1] for r in range_rows)/first_price-1)*100, 3) if range_rows else None
    d["observed_high_pct"] = round((max(r[2] for r in range_rows)/first_price-1)*100, 3) if range_rows else None
    for minutes in (5, 15, 30, 60):
        # First completed candle AT/AFTER milestone, at most 2 min late.
        candidates = [r for r in rows if first_ts + minutes*60 <= r[0] <= first_ts+(minutes+2)*60]
        d[f"return_{minutes}m_pct"] = round((candidates[0][3]/first_price-1)*100, 3) if candidates else None
    d["observed_bars"] = len(rows)
    return d

def existing_sqlite_login():
    """Convert the original bot's authorized SQLite session to an in-memory
    Telethon StringSession. This function reads SQLite in read-only mode and
    never copies or prints its authentication key or changes .env.
    """
    import base64
    import ipaddress
    import struct
    from urllib.parse import quote

    path = Path.home() / "uditbot" / "data" / "tg_userbot.session"
    if not path.is_file():
        return ""
    try:
        uri = "file:" + quote(str(path), safe="/") + "?mode=ro"
        with sqlite3.connect(uri, uri=True, timeout=5) as conn:
            row = conn.execute(
                "SELECT dc_id, server_address, port, auth_key "
                "FROM sessions LIMIT 1"
            ).fetchone()
        if not row:
            raise ValueError("empty sessions table")
        dc, address, port, key = row
        key_bytes = bytes(key)
        if len(key_bytes) != 256:
            raise ValueError("invalid session authentication key length")
        packed_ip = ipaddress.ip_address(str(address)).packed
        fmt = ">B4sH256s" if len(packed_ip) == 4 else ">B16sH256s"
        encoded = struct.pack(fmt, int(dc), packed_ip, int(port), key_bytes)
        return "1" + base64.urlsafe_b64encode(encoded).decode("ascii")
    except (OSError, sqlite3.Error, ValueError, TypeError, struct.error) as exc:
        raise RuntimeError(
            "Existing ~/uditbot/data/tg_userbot.session could not be read; "
            "credentials were not changed"
        ) from exc


def require_trading_inactive():
    """Fail closed before sharing an existing Telegram authorization."""
    import subprocess
    try:
        result = subprocess.run(
            ["systemctl", "is-active", "trading_bot.service"],
            capture_output=True, text=True, timeout=5, check=False,
        )
    except (OSError, subprocess.TimeoutExpired) as e:
        raise RuntimeError("Cannot verify trading bot state; refusing session reuse") from e
    status = result.stdout.strip()
    if status not in ("inactive", "failed"):
        raise RuntimeError(
            "Cannot reuse Telegram login while trading_bot.service is active or its state is uncertain: "
            + repr(status)
        )

import asyncio
import re
import sqlite3
from pathlib import Path
from zoneinfo import ZoneInfo
from dotenv import dotenv_values
from telethon import TelegramClient, events
from telethon.sessions import StringSession

ROOT = Path.home() / "uditbot"
STORE = Path.home() / "resultbot-data"
DB_PATH = STORE / "alerts.sqlite3"
IST = ZoneInfo("Asia/Kolkata")

PATTERN = re.compile(
    r"^\s*#([A-Z][A-Z0-9&_.-]{0,24})\s*[-–—]\s*"
    r"(?:🏆\s*🔥\s*)?Excellent Results\b",
    re.IGNORECASE
)

def save_alert(db, channel_id, msg):
    raw = msg.raw_text or ""
    match = PATTERN.search(raw)
    if not match:
        return False

    cur = db.execute("""
        INSERT OR IGNORE INTO alerts
        (channel_id, message_id, symbol, telegram_time_utc, raw_text)
        VALUES (?, ?, ?, ?, ?)
    """, (
        channel_id, msg.id, match.group(1).upper(),
        msg.date.isoformat(), raw
    ))

    if cur.rowcount:
        print(
            "SAVED", match.group(1).upper(),
            msg.date.astimezone(IST).strftime("%Y-%m-%d %H:%M IST"),
            "message", msg.id,
            flush=True
        )
        return True
    return False

async def main():
    env = dotenv_values(ROOT / ".env")
    api_id = int(env.get("TG_API_ID") or 0)
    api_hash = env.get("TG_API_HASH") or ""

    data_dir = Path(env.get("DATA_DIR") or "data")
    if not data_dir.is_absolute():
        data_dir = ROOT / data_dir

    session_file = data_dir / "tg_userbot.session"

    if not api_id or not api_hash:
        raise RuntimeError("Missing Telegram API settings")

    if session_file.is_file():
        session = str(session_file)
    elif env.get("TG_SESSION_STRING"):
        session = StringSession(env["TG_SESSION_STRING"])
    else:
        raise RuntimeError("No Telegram session found")

    db = sqlite3.connect(DB_PATH, timeout=30)
    db.execute("PRAGMA journal_mode=WAL")
    db.execute("PRAGMA busy_timeout=30000")

    db.execute("""
        CREATE TABLE IF NOT EXISTS alerts (
            channel_id INTEGER NOT NULL,
            message_id INTEGER NOT NULL,
            symbol TEXT NOT NULL,
            telegram_time_utc TEXT NOT NULL,
            raw_text TEXT NOT NULL,
            PRIMARY KEY(channel_id, message_id)
        )
    """)

    db.execute("""
        CREATE TABLE IF NOT EXISTS listener_state (
            channel_id INTEGER PRIMARY KEY,
            last_message_id INTEGER NOT NULL
        )
    """)
    db.commit()

    client = TelegramClient(
        session, api_id, api_hash, auto_reconnect=True
    )
    await client.connect()

    try:
        if not await client.is_user_authorized():
            raise RuntimeError("Telegram login not authorized")

        channel = await client.get_entity("earnings_pulse")

        print(
            f"ResultBot connected: {channel.title} ({channel.id})",
            flush=True
        )

        async def catch_up():
            row = db.execute(
                "SELECT last_message_id FROM listener_state "
                "WHERE channel_id=?", (channel.id,)
            ).fetchone()

            if row:
                last_id = row[0]
            else:
                last_id = db.execute(
                    "SELECT COALESCE(MAX(message_id), 0) "
                    "FROM alerts WHERE channel_id=?",
                    (channel.id,)
                ).fetchone()[0]

            scanned = new = 0

            async for msg in client.iter_messages(
                channel, min_id=last_id, reverse=True
            ):
                with db:
                    new += save_alert(db, channel.id, msg)
                    db.execute("""
                        INSERT INTO listener_state
                        (channel_id, last_message_id)
                        VALUES (?, ?)
                        ON CONFLICT(channel_id) DO UPDATE SET
                        last_message_id = MAX(
                            listener_state.last_message_id,
                            excluded.last_message_id
                        )
                    """, (channel.id, msg.id))
                scanned += 1

            print(
                f"Catch-up: {scanned} scanned, "
                f"{new} new alerts",
                flush=True
            )

        @client.on(events.NewMessage(chats=channel))
        async def on_message(event):
            with db:
                save_alert(db, channel.id, event.message)

        async def periodic_catchup():
            while True:
                await asyncio.sleep(300)
                try:
                    await catch_up()
                except Exception as exc:
                    print(
                        f"Catch-up retry: {type(exc).__name__}: {exc}",
                        flush=True
                    )

        await catch_up()
        task = asyncio.create_task(periodic_catchup())

        try:
            print("RESULTBOT LISTENING 24/7", flush=True)
            await client.run_until_disconnected()
        finally:
            task.cancel()
            try:
                await task
            except asyncio.CancelledError:
                pass
    finally:
        await client.disconnect()
        db.close()

if __name__ == "__main__":
    asyncio.run(main())

def valid(d):
    if d["down_mode"] not in ("CUT","AVERAGE","STOP_ONLY") or d["up_mode"] not in ("TAKE_PROFIT","ADD_UP","TARGET_ONLY"):
        raise ValueError("Unknown position-management mode")
    for k in ("stop_pct","target_pct","initial_budget_inr","extra_budget_inr","total_buy_cap_inr"):
        if not isinstance(d[k],(int,float)) or isinstance(d[k],bool) or not math.isfinite(d[k]) or d[k]<=0:
            raise ValueError(k+" must be a positive finite number")
    if d["stop_pct"]>=50 or d["target_pct"]>100 or d["initial_budget_inr"]>d["total_buy_cap_inr"]:
        raise ValueError("Invalid price exits or budget cap")
    for key in ("average_down","add_up"):
        xs=d[key]
        if len(xs)>5 or any(not isinstance(x,(int,float)) or not 0<x<100 for x in xs) or sorted(set(xs))!=xs:
            raise ValueError(key+": specify 0-5 ascending distinct percentages")
    for key in ("average_down","add_up"):
        amounts = d[key+"_amounts_inr"]
        if (not isinstance(amounts,list) or len(amounts)!=len(d[key]) or
            any(not isinstance(x,(int,float)) or isinstance(x,bool) or not math.isfinite(x) or x<=0 for x in amounts)):
            raise ValueError(key+" budgets: one positive INR amount for each price trigger")
    for key in ("cut_down","take_profit"):
        xs=d[key]
        if len(xs)>5 or any(not isinstance(x,list) or len(x)!=2 for x in xs):
            raise ValueError(key+": specify up to 5 trigger:cumulative-exit pairs")
        if not all(0<x[0]<100 and 0<x[1]<100 for x in xs):
            raise ValueError(key+": triggers and exit fractions must be in 0..100")
        if [x[0] for x in xs]!=sorted(set(x[0] for x in xs)) or [x[1] for x in xs]!=sorted(set(x[1] for x in xs)):
            raise ValueError(key+": triggers and cumulative sold percentages must increase")
    down = d["average_down"] if d["down_mode"]=="AVERAGE" else [p for p,q in d["cut_down"]] if d["down_mode"]=="CUT" else []
    up = d["add_up"] if d["up_mode"]=="ADD_UP" else [p for p,q in d["take_profit"]] if d["up_mode"]=="TAKE_PROFIT" else []

def cmd_refresh_token(cfg: AppConfig, stop_mgr: EmergencyStopManager) -> bool:
    """Generates a fresh 24-hour Dhan access token using Client ID, PIN, and TOTP secret via shared lock."""
    from src.broker.dhan_client import DhanClient

    print("=======================================================")
    print("         DHAN AUTOMATED TOTP TOKEN REFRESH            ")
    print("=======================================================")
    print(f"[*] Dhan Client ID    : {cfg.dhan_client_id or 'Not Configured'}")
    print(f"[*] Dhan PIN          : {'Configured' if cfg.dhan_pin else 'MISSING'}")
    print(f"[*] Dhan TOTP Secret  : {'Configured' if cfg.dhan_totp_secret else 'MISSING'}")

    if not cfg.dhan_client_id or not cfg.dhan_pin or not cfg.dhan_totp_secret:
        print("\n❌ Missing credentials in .env.")
        print("Please ensure DHAN_CLIENT_ID, DHAN_PIN, and DHAN_TOTP_SECRET are set.")
        return False

    from src.broker.dhan_token import read_token, write_token_atomic
    token_path = os.path.expanduser(cfg.dhan_token_file)
    prev_token = read_token(token_path) if os.path.exists(token_path) else ""

    client = DhanClient(cfg, stop_manager=stop_mgr)
    ok, token, msg = client.generate_token_with_totp(force=True)
    if ok and token:
        ok_f, bal, msg_f = client.verify_token_authenticated()
        if not ok_f:
            print(f"\n❌ Post-refresh broker verification failed: {msg_f}")
            if prev_token and prev_token != token:
                write_token_atomic(prev_token, token_path)
                client.token = prev_token
                try:
                    client._token_mtime = os.path.getmtime(token_path)
                except Exception:
                    pass
                ok_prev, bal_prev, msg_prev = client.verify_token_authenticated()
                if ok_prev:
                    print(
                        f"[*] RECOVERY_VERIFIED: Restored previous token and confirmed broker authentication "
                        f"(Balance: INR {bal_prev:,.2f})."
                    )
                else:
                    print(
                        f"[!] RECOVERY_UNVERIFIED: Previous token is no longer valid after new TOTP attempt "
                        f"({msg_prev}). Broker authentication remains UNVERIFIED!"
                    )
            else:
                print("[!] No distinct previous token available for recovery; broker authentication remains UNVERIFIED.")
            print("=======================================================")
            return False

        print("\n✅ Successfully generated fresh Dhan access token via TOTP!")
        print(f"[*] Saved to token file: {cfg.dhan_token_file} (Single authoritative token source)")
        print(f"[*] Available Balance : INR {bal:,.2f}")
        print("=======================================================")
        return True
    else:
        print(f"\n❌ Failed to generate token: {msg}")
        print("=======================================================")
        return False


def cmd_daily_health_check(cfg: AppConfig, db: JournalDB, stop_mgr: EmergencyStopManager) -> int:
    """
    V12-24: Executes the scheduled 08:00 IST daily token refresh and health probe invoked by
    `dhan-token-refresh.service`. Returns exit code 0 on success, 1 on failure.
    """
    from src.broker.dhan_client import DhanClient
    from src.instruments.resolver import InstrumentResolver

    print("=======================================================")
    print("       AUDIT BOT 08:00 IST DAILY HEALTH CHECK          ")
    print("=======================================================")

    tok_ok = cmd_refresh_token(cfg, stop_mgr)
    client = DhanClient(cfg, stop_manager=stop_mgr)
    ok_funds, avail_bal, funds_msg = client.get_fund_limits(retry=False)

    scrip_path = os.path.join(cfg.data_dir, "scrip_master.csv")
    resolver = InstrumentResolver()
    master_ok = resolver.load_dhan_master(
        scrip_path,
        max_cache_age_sec=cfg.scrip_master_max_age_sec,
        allow_network_download=True,
    )

    overall_ok = bool(tok_ok and ok_funds)
    summary = (

    if overall_ok:
        print(f"✅ Daily health check passed: {summary}")
        return 0
    print(f"❌ Daily health check failed: {summary}")
    return 1

def cmd_rules(db: JournalDB, sub_args: list) -> int:
    from src.parsers.adaptive_rules import AdaptiveRuleManager
    mgr = AdaptiveRuleManager(db)
    sub = (sub_args[0].lower() if sub_args else "list")
    if sub in ("list", "ls"):
        rules = db.list_parser_rules()
        print("=======================================================")
        print("          AUDIT BOT ADAPTIVE PARSER RULES              ")
        print("=======================================================")
        if not rules:
            print("  No adaptive parser rules registered.")
        for r in rules:
            print(f"  [{r['state']}] {r['rule_id']} (v{r['version']}, route={r['route']}) — {r['pattern']}")
        print("=======================================================")
        return 0
    elif sub == "versions" and len(sub_args) >= 2:
        rule_id = sub_args[1]
        vers = db.get_parser_rule_versions(rule_id)
        for v in vers:
            print(f"  {v['rule_id']} v{v['version']} [{v['state']}] approved_by={v['approved_by']!r} pattern={v['pattern']}")
        return 0
    elif sub == "approve" and len(sub_args) >= 2:
        rule_id = sub_args[1]
        reviewer = sub_args[2] if len(sub_args) >= 3 else "Udit"
        notes = " ".join(sub_args[3:]) if len(sub_args) >= 4 else ""
        ok, msg = mgr.approve_rule(rule_id, reviewer=reviewer, notes=notes)
        print(f"{'✅' if ok else '❌'} {msg}")
        return 0 if ok else 1
    elif sub == "activate" and len(sub_args) >= 2:
        rule_id = sub_args[1]
        actor = sub_args[2] if len(sub_args) >= 3 else "Udit"
        ok, msg = mgr.activate_rule(rule_id, actor=actor)
        print(f"{'✅' if ok else '❌'} {msg}")
        return 0 if ok else 1
    elif sub in ("deactivate", "rollback") and len(sub_args) >= 2:
        rule_id = sub_args[1]
        actor = sub_args[2] if len(sub_args) >= 3 else "Udit"
        reason = " ".join(sub_args[3:]) if len(sub_args) >= 4 else "Operator CLI deactivation"
        ok, msg = mgr.deactivate_rule(rule_id, actor=actor, reason=reason)
        print(f"{'✅' if ok else '❌'} {msg}")
        return 0 if ok else 1
    elif sub == "restore" and len(sub_args) >= 3 and sub_args[2].isdigit():
        rule_id = sub_args[1]
        ver_num = int(sub_args[2])
        actor = sub_args[3] if len(sub_args) >= 4 else "Udit"
        ok, msg, _ = mgr.restore_rule_version(rule_id, target_version=ver_num, actor=actor, activate=False)
        print(f"{'✅' if ok else '❌'} {msg}")
        if ok:
            print("Rule restored in APPROVED state only; run `bot rules activate <ID> <actor>` separately to enable it.")
        return 0 if ok else 1
    else:
        print("Usage: bot rules [list | versions <ID> | approve <ID> [reviewer] | activate <ID> [actor] | deactivate <ID> [actor] [reason] | restore <ID> <version> [actor]]")
        return 2


def cmd_service_control(cfg: AppConfig, action: str, args: list) -> int:
    """B10 & B12: Owner-authorized service start/stop/restart/status command with safety and control-state tracking."""
    from src.engine.config_manager import ConfigChangeTracker
    tracker = ConfigChangeTracker(data_dir=cfg.data_dir)
    confirm_live = ("--yes" in args or "-y" in args)
    clean_args = [a for a in args if a not in ("--yes", "-y")]
    reason = " ".join(clean_args).strip() or f"Operator CLI {action}"

    if action in ("start", "restart") and cfg.trading_mode.value == "LIVE" and not confirm_live:
        ans = input(f"You are about to {action} AuditBot in LIVE mode. Real Dhan orders may be placed. Continue? (yes/no): ").strip().lower()
        if ans not in ("yes", "y"):
            print(f"Aborted LIVE service {action}.")
            return 1
        confirm_live = True

    env_p = resolve_env_path(".env")
    ro_broker = tracker.get_readonly_broker(env_path=env_p) if action != "status" else None
    res = tracker.control_service(
        action=action,
        reason=reason,
        confirm_live=confirm_live,
        env_path=env_p,
        stop_state_path=os.path.join(cfg.data_dir, "emergency_stop.json"),
        db_path=cfg.journal_db_path,
        broker=ro_broker,
    )
    msg = res.get("message") or res.get("error") or f"Service {action} completed"
    if res.get("ok"):
        print(f"✅ {msg} ({res.get('service_state_label', res.get('service_state'))})")
        return 0
    if res.get("deferred"):
        print(f"⏳ {msg}")
        return 0
    print(f"❌ {msg}")
    return 1
